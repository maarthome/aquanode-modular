# Linux build of Aquanode Modular (added in the maarthome fork; upstream builds with Projucer).
#
# The list of source files is read from the upstream .jucer project at configure time, so
# files added or removed upstream are picked up automatically after syncing the fork.
#
# Usually driven by ./build_pi.sh, which also installs the packages and fetches JUCE.

cmake_minimum_required(VERSION 3.22)

# ── Read name, version and source list from the Projucer project ─────────────────────────
set(AQ_JUCER "${CMAKE_CURRENT_SOURCE_DIR}/Aquanode_Modular.jucer")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${AQ_JUCER}")
file(READ "${AQ_JUCER}" AQ_JUCER_TEXT)

string(REGEX MATCH "<JUCERPROJECT[^>]*>" AQ_HEADER "${AQ_JUCER_TEXT}")
string(REGEX MATCH " version=\"([0-9.]+)\"" _ "${AQ_HEADER}")
set(AQ_VERSION "${CMAKE_MATCH_1}")
if(NOT AQ_VERSION)
    message(FATAL_ERROR "Could not read the version from ${AQ_JUCER}")
endif()

string(REGEX MATCHALL "<FILE [^>]*>" AQ_FILE_TAGS "${AQ_JUCER_TEXT}")
set(AQ_SOURCES "")
set(AQ_RESOURCES "")
foreach(tag IN LISTS AQ_FILE_TAGS)
    if(NOT tag MATCHES " file=\"([^\"]+)\"")
        continue()
    endif()
    set(path "${CMAKE_CURRENT_SOURCE_DIR}/${CMAKE_MATCH_1}")
    if(tag MATCHES " compile=\"1\"")
        list(APPEND AQ_SOURCES "${path}")
    elseif(tag MATCHES " resource=\"1\"")
        list(APPEND AQ_RESOURCES "${path}")
    endif()
endforeach()
list(LENGTH AQ_SOURCES AQ_SOURCE_COUNT)
if(AQ_SOURCE_COUNT EQUAL 0)
    message(FATAL_ERROR "No source files found in ${AQ_JUCER}")
endif()
foreach(path IN LISTS AQ_SOURCES AQ_RESOURCES)
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Listed in the .jucer but missing on disk (check upper/lower case): ${path}")
    endif()
endforeach()

project(AquanodeModular VERSION "${AQ_VERSION}" LANGUAGES C CXX)
message(STATUS "Aquanode Modular ${AQ_VERSION}: ${AQ_SOURCE_COUNT} source files from the .jucer")

# aq_add_plugin(<target> <product name> <4-char code> <lv2 uri> <synth?> <formats...>)
function(aq_add_plugin target product code uri is_synth)
    set(formats ${ARGN})
    if(AQ_BUILD_VST3)
        list(APPEND formats VST3)
        list(REMOVE_DUPLICATES formats)
    endif()

    if(is_synth)
        set(kind IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT TRUE)
    else()
        set(kind IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE VST3_CATEGORIES Fx)
    endif()

    juce_add_plugin(${target}
        PRODUCT_NAME             "${product}"
        VERSION                  "${AQ_VERSION}"
        COMPANY_NAME             "aquanodemusic"
        COMPANY_EMAIL            "node.aqua@gmail.com"
        PLUGIN_MANUFACTURER_CODE Aqnd
        PLUGIN_CODE              ${code}
        LV2URI                   "${uri}"
        FORMATS                  ${formats}
        ${kind}
        COPY_PLUGIN_AFTER_BUILD  FALSE)

    set(AQ_PRODUCT_NAME "${product}")
    configure_file("${AQ_GENERATED}/JuceHeader.h.in" "${AQ_GENERATED}/${target}/JuceHeader.h" @ONLY)

    target_sources(${target} PRIVATE ${AQ_SOURCES})
    target_include_directories(${target} PRIVATE "${AQ_GENERATED}/${target}")

    target_compile_definitions(${target} PUBLIC ${AQ_COMMON_DEFINITIONS})
    if(NOT is_synth)
        target_compile_definitions(${target} PUBLIC AQUANODE_FX_BUILD=1)
    endif()

    # The code base is developed with MSVC; keep GCC quiet about style, loud about errors.
    target_compile_options(${target} PRIVATE -w)

    target_link_libraries(${target}
        PRIVATE
            ${AQ_LINK_MODULES}
        PUBLIC
            juce::juce_recommended_config_flags)
endfunction()

# ── Stripped plugin: the engine plus only the modules one patch uses, no editor window ────
# Driven by ./build_stripped.sh. tools/make_stripped.py writes AQ_STRIPPED_DIR.
#
# Only the engine is compiled here (with AQUANODE_STRIPPED=1). The modules come out of the
# full build's library: the generated registry names the wanted modules, and the linker
# takes just those objects, and what they need, from libAquanode Modular_SharedCode.a.
if(AQ_STRIPPED_DIR)
    include("${AQ_STRIPPED_DIR}/stripped.cmake")
    string(MAKE_C_IDENTIFIER "AquanodeStripped_${AQS_SLUG}" AQS_TARGET)

    if(AQS_IS_FX)
        set(aqs_kind IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE)
    else()
        set(aqs_kind IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT TRUE)
    endif()

    juce_add_plugin(${AQS_TARGET}
        PRODUCT_NAME             "${AQS_NAME}"
        VERSION                  "${AQ_VERSION}"
        COMPANY_NAME             "aquanodemusic"
        PLUGIN_MANUFACTURER_CODE Aqnd
        PLUGIN_CODE              ${AQS_CODE}
        LV2URI                   "https://github.com/maarthome/aquanode-modular/stripped/${AQS_SLUG}"
        FORMATS                  LV2
        ${aqs_kind}
        COPY_PLUGIN_AFTER_BUILD  FALSE)

    set(AQ_PRODUCT_NAME "${AQS_NAME}")
    configure_file("${AQ_GENERATED}/JuceHeader.h.in" "${AQ_GENERATED}/${AQS_TARGET}/JuceHeader.h" @ONLY)

    list(TRANSFORM AQS_SOURCES PREPEND "${CMAKE_CURRENT_SOURCE_DIR}/")
    target_sources(${AQS_TARGET} PRIVATE
        ${AQS_SOURCES}
        "${AQ_STRIPPED_DIR}/StrippedRegistry.cpp"
        "${AQ_STRIPPED_DIR}/StrippedStart.cpp")
    target_include_directories(${AQS_TARGET} PRIVATE
        "${AQ_GENERATED}/${AQS_TARGET}"
        "${CMAKE_CURRENT_SOURCE_DIR}")

    # Same settings as the full build, so its library fits; plus the stripped switch.
    target_compile_definitions(${AQS_TARGET} PUBLIC ${AQ_COMMON_DEFINITIONS} AQUANODE_STRIPPED=1)
    if(AQS_IS_FX)
        target_compile_definitions(${AQS_TARGET} PUBLIC AQUANODE_FX_BUILD=1)
    endif()
    target_compile_options(${AQS_TARGET} PRIVATE -w -ffunction-sections -fdata-sections)

    target_link_libraries(${AQS_TARGET}
        PRIVATE
            ${AQ_LINK_MODULES}
            "$<TARGET_FILE:AquanodeModular>"
        PUBLIC
            juce::juce_recommended_config_flags)
    add_dependencies(${AQS_TARGET} AquanodeModular)
    target_link_options(${AQS_TARGET}_LV2 PRIVATE -Wl,--gc-sections)

    # fixed names for build_stripped.sh
    add_custom_target(aq_stripped DEPENDS ${AQS_TARGET}_LV2)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/aq_stripped_artefacts.txt"
        "${CMAKE_CURRENT_BINARY_DIR}/${AQS_TARGET}_artefacts/${CMAKE_BUILD_TYPE}/LV2\n")
endif()
